function HousemateLoginForm({ housemates, setCurrentUser, showAlert }) {
  const [nameInput, setNameInput] = useState('');
  const [codeInput, setCodeInput] = useState('');

  const handleLogin = (e) => {
    e.preventDefault();
    const hm = housemates.find(h => h.name.toLowerCase().trim() === nameInput.toLowerCase().trim() && h.code === codeInput.trim());
    if (!hm) {
      showAlert('Invalid Housemate Name or Access Code.');
      return;
    }
    if (hm.status === 'Evicted' || hm.status === 'Suspended') {
      showAlert('Your account has been evicted or suspended. Privileges revoked.');

    setCodeInput('');

function HousematePortalContent({ housemates, currentUser, nominations, setNominations, activeRound, syncToCloud, showAlert }) {
  const hm = currentUser.housemate;
  const weekKey = `week_${activeRound.week}`;
  const myNomination = nominations[weekKey]?.[hm.id];

  const [firstNominee, setFirstNominee] = useState('');
  const [firstReason, setFirstReason] = useState('');
  const [secondNominee, setSecondNominee] = useState('');
  const [secondReason, setSecondReason] = useState('');

  const handleNominationSubmit = async (e) => {
    e.preventDefault();
    if (firstNominee === secondNominee) {
      showAlert('You cannot nominate the same housemate twice.');
      return;
    }
    if (firstNominee === hm.id || secondNominee === hm.id) {
      showAlert('You cannot nominate yourself.');
      return;
    }
    if (!firstReason.trim() || !secondReason.trim()) {
      showAlert('Both nomination reasons are required.');
      return;
    }

    const currentSubmissions = nominations[weekKey] || {};
    if (currentSubmissions[hm.id]) {
      showAlert('You have already submitted nominations for this week.');
      return;
    }

    const updatedNoms = {
      ...nominations,
      [weekKey]: {
        ...currentSubmissions,
        [hm.id]: {
          first: { to: firstNominee, reason: firstReason, points: 3 },
          second: { to: secondNominee, reason: secondReason, points: 1 }
        }
      }
    };

    setNominations(updatedNoms);
    await syncToCloud({ nominations: updatedNoms });

    showAlert('Nominations submitted and synced to the cloud vault!');
  };

  return (
    <div className="space-y-6">
      <div className="border-b border-slate-800 pb-4">
        <h3 className="text-lg font-black text-[#D4AF37]">Weekly Online Nominations (Week {activeRound.week})</h3>
        <p className="text-xs text-slate-300">Nominate 2 different housemates (3 points for 1st, 1 point for 2nd with reasons).</p>
      </div>

      {myNomination ? (
        <div className="bg-[#1A1A24] p-6 rounded-2xl border border-[#D4AF37]/50 space-y-4">
          <div className="text-xs font-black text-emerald-400 flex items-center space-x-2">
            <CheckCircle className="w-5 h-5" />
            <span>Nominations Successfully Submitted & Synced</span>
          </div>
          <div className="grid grid-cols-1 sm:grid-cols-2 gap-4 text-sm font-medium">
            <div className="bg-[#0D0D11] p-4 rounded-xl border border-slate-800 space-y-1">
              <span className="text-xs text-[#D4AF37] font-black">1st Nominee (3 pts): {myNomination.first.to}</span>
              <p className="text-xs text-slate-300">Reason: {myNomination.first.reason}</p>
            </div>
            <div className="bg-[#0D0D11] p-4 rounded-xl border border-slate-800 space-y-1">
              <span className="text-xs text-[#D4AF37] font-black">2nd Nominee (1 pt): {myNomination.second.to}</span>
              <p className="text-xs text-slate-300">Reason: {myNomination.second.reason}</p>
            </div>
          </div>
        </div>
      ) : (
        <form onSubmit={handleNominationSubmit} className="space-y-6">
          <div className="grid grid-cols-1 md:grid-cols-2 gap-6">
            <div className="space-y-3 bg-[#1A1A24] p-5 rounded-2xl border border-slate-800">
              <label className="text-xs font-black uppercase text-[#D4AF37]">First Nominee (3 Points)</label>
              <select 
                value={firstNominee} 
                onChange={e => setFirstNominee(e.target.value)} 
                className="w-full bg-[#0D0D11] border border-slate-800 rounded-xl px-4 py-3 text-sm text-white font-bold"
                required
              >
                <option value="">Select Housemate</option>
                {housemates.filter(h => h.id !== hm.id && h.status === 'Active').map(h => (
                  <option key={h.id} value={h.id}>{h.name} ({h.id})</option>
                ))}
              </select>
              <textarea 
                value={firstReason} 
                onChange={e => setFirstReason(e.target.value)} 
                placeholder="Enter nomination reason..." 
                className="w-full bg-[#0D0D11] border border-slate-800 rounded-xl p-4 text-sm text-white h-24 focus:outline-none focus:border-[#D4AF37]"

              />
            </div>

            <div className="space-y-3 bg-[#1A1A24] p-5 rounded-2xl border border-slate-800">
              <label className="text-xs font-black uppercase text-[#D4AF37]">Second Nominee (1 Point)</label>

                className="w-full bg-[#0D0D11] border border-slate-800 rounded-xl px-4 py-3 text-sm text-white font-bold"
                required
              >
                <option value="">Select Housemate</option>
                {housemates.filter(h => h.id !== hm.id && h.status === 'Active').map(h => (

function VotingView({ activeRound, currentWeekVotes, votes, setVotes, resultsConfig, syncToCloud, showAlert, goBack }) {
  const [selectedNomineeId, setSelectedNomineeId] = useState('');
  const [voteType, setVoteType] = useState('save'); 

  const nomineeList = activeRound.nominees || [];

  const handleVoteSubmission = async (e) => {
    e.preventDefault();
    if (resultsConfig.votingManuallyClosed) {
      showAlert('Voting has been manually closed by the administrator.');
      return;
    }
    if (!selectedNomineeId) {
      showAlert('Please select a nominee.');
      return;
    }

    const nomObj = nomineeList.find(n => n.id === selectedNomineeId);
    const nomName = nomObj ? nomObj.label : selectedNomineeId;

    const weekNum = activeRound.week;
    const weekVotes = votes[weekNum] || { save: {}, evict: {} };
    const category = voteType === 'save' ? 'save' : 'evict';
    const currentCount = weekVotes[category][selectedNomineeId] || 0;

    const updatedVotes = {
      ...votes,
      [weekNum]: {
        ...weekVotes,
        [category]: {
          ...weekVotes[category],
          [selectedNomineeId]: currentCount + 1
        }
      }
    };

    setVotes(updatedVotes);
    await syncToCloud({ votes: updatedVotes });

    showAlert(`Successfully cast your Vote to ${voteType === 'save' ? 'SAVE' : 'EVICT'} for ${nomName}! Synced live to cloud.`);
    setSelectedNomineeId('');
  };

  return (
    <div className="max-w-4xl mx-auto px-4 sm:px-6 lg:px-8 py-12 space-y-10 animate-fadeIn">
      <div className="text-center space-y-3 border-b border-slate-800 pb-6">

function ResultsView({ activeRound, currentWeekVotes, resultsConfig, goBack }) {
  const nomineeList = activeRound.nominees || [];
  
  const saveVotesMap = currentWeekVotes.save || {};
  const evictVotesMap = currentWeekVotes.evict || {};

  const totalSaveVotes = Object.values(saveVotesMap).reduce((a, b) => a + b, 0) || 1;
  const totalEvictVotes = Object.values(evictVotesMap).reduce((a, b) => a + b, 0) || 1;

  const calculatedResults = nomineeList.map((nom, idx) => {
    const saveCount = saveVotesMap[nom.id] || 0;
    const evictCount = evictVotesMap[nom.id] || 0;

    const savePct = (saveCount / totalSaveVotes) * 100;
    const evictPct = (evictCount / totalEvictVotes) * 100;
    const finalScore = savePct - evictPct;

    return {
      id: nom.id,
      name: nom.label,
      photo: nom.photo || '',
      letter: String.fromCharCode(65 + idx),
      savePct,
      evictPct,
      finalScore
    };
  }).sort((a, b) => b.savePct - a.savePct);

  return (
    <div className="max-w-5xl mx-auto px-4 sm:px-6 lg:px-8 py-12 space-y-12 animate-fadeIn">
      <div className="text-center space-y-2 border-b border-slate-800 pb-6">
        <div className="inline-flex items-center space-x-2 text-black bg-[#D4AF37] px-4 py-1.5 rounded-full text-xs font-black uppercase tracking-widest shadow-lg">
          <Trophy className="w-4 h-4" />
          <span>Official Tally & Eviction Hall • Week {activeRound.week}</span>
        </div>
        <h1 className="text-3xl sm:text-5xl font-black text-white">RESULTS & EVICTION</h1>
        <p className="text-sm text-slate-300 font-medium">
          Official formula: Final Net Score = Save % − Evict %
        </p>
      </div>

      {/* PBB GEN 11 TALLY GRAPHIC LAYOUT */}
      <div className="bg-[#14141C] border-4 border-[#D4AF37] rounded-3xl p-6 sm:p-10 space-y-8 shadow-2xl relative overflow-hidden">
        <div className="flex flex-col sm:flex-row items-center justify-between border-b border-slate-800 pb-6 gap-4">
          <div>
            <div className="text-xs tracking-widest text-[#D4AF37] font-black uppercase">Broadcast Graphic</div>
            <h3 className="text-2xl sm:text-3xl font-black text-white flex items-center space-x-3">
              <Radio className="w-7 h-7 text-[#D4AF37] animate-pulse" />
              <span>TALLY OF VOTES (WEEK {activeRound.week})</span>
            </h3>
          </div>
          <span className="text-xs font-mono font-bold bg-[#1A1A24] text-emerald-400 px-4 py-1.5 rounded-full border border-emerald-500/40">
            {resultsConfig.showPartial ? '🟢 Live Sync Public' : '🔒 Hidden by Admin'}
          </span>
        </div>

        {!resultsConfig.showPartial ? (
          <div className="text-center py-16 text-slate-400 font-bold text-sm">
            The partial tally board is currently hidden by administrator controls.
          </div>
        ) : (
          <div className="space-y-6 max-w-3xl mx-auto">
            <div className="flex justify-between items-center text-xs font-black uppercase text-slate-300 px-4">
              <span className="text-emerald-400">Vote to Save</span>
              <span className="text-[#D4AF37]">Nominee</span>
              <span className="text-red-400">Vote to Evict</span>
            </div>

            {calculatedResults.map((res) => {
              const badgeColors = { 'A': 'bg-blue-600', 'B': 'bg-amber-600', 'C': 'bg-red-600', 'D': 'bg-emerald-600' };
              const colorClass = badgeColors[res.letter] || 'bg-blue-600';

              return (
                <div key={res.id} className="bg-[#0D0D11] border-2 border-slate-800 rounded-2xl p-4 sm:p-5 flex items-center justify-between shadow-xl gap-2 sm:gap-4 relative overflow-hidden">
                  {/* Left: Save Percentage Bar */}
                  <div className="flex-1 flex items-center justify-end space-x-3">
                    <span className="text-base sm:text-xl font-black text-emerald-400 font-mono">{res.savePct.toFixed(2)}%</span>
                    <div className="hidden sm:block w-32 bg-[#1A1A24] h-3 rounded-full overflow-hidden border border-emerald-500/40">
                      <div className="bg-emerald-400 h-full rounded-full transition-all duration-1000" style={{ width: `${Math.min(res.savePct, 100)}%` }}></div>
                    </div>
                  </div>

                  {/* Center: Nominee Badge with Avatar */}
                  <div className={`px-4 sm:px-6 py-2.5 rounded-2xl ${colorClass} border-2 border-[#D4AF37] flex items-center space-x-3 shadow-lg transform hover:scale-105 transition`}>
                    <span className="text-xl sm:text-2xl font-black text-white">{res.letter}</span>
                    <div className="w-10 h-10 rounded-full overflow-hidden border-2 border-white/80 shadow">
                      <img src={res.photo || 'https://placehold.co/200x200/0D0D11/D4AF37?text=Nominee'} alt={res.name} className="w-full h-full object-cover" onError={(e)=>{e.target.src='https://placehold.co/200x200/0D0D11/D4AF37?text=Nominee'}} />
                    </div>
                  </div>

                  {/* Right: Evict Percentage Bar */}
                  <div className="flex-1 flex items-center justify-start space-x-3">
                    <div className="hidden sm:block w-32 bg-[#1A1A24] h-3 rounded-full overflow-hidden border border-red-500/40">
                      <div className="bg-red-500 h-full rounded-full transition-all duration-1000" style={{ width: `${Math.min(res.evictPct, 100)}%` }}></div>
                    </div>
                    <span className="text-base sm:text-xl font-black text-red-400 font-mono">{res.evictPct.toFixed(2)}%</span>
                  </div>
                </div>

            })}
          </div>
        )}
      </div>

      {/* Final Results & Rankings Card */}

          <h3 className="text-xl font-black text-[#D4AF37] flex items-center space-x-2">
            <Award className="w-6 h-6" />
            <span>FINAL RESULTS & RANKINGS</span>
          </h3>
          <span className="text-xs text-slate-300 font-mono font-bold bg-[#1A1A24] px-3 py-1 rounded-full border border-[#D4AF37]/30">
            {resultsConfig.showFinal ? '🟢 Published Live' : '🔒 Awaiting Admin Publication'}
          </span>
        </div>

        {!resultsConfig.showFinal ? (
          <div className="text-center py-12 text-slate-400 space-y-3 font-bold">
            <Lock className="w-10 h-10 text-[#D4AF37] mx-auto animate-bounce" />
            <p className="text-sm">Final results have not yet been published by the administrator.</p>
          </div>
        ) : (
          <div className="space-y-4">
            {calculatedResults.map((res, index) => {
              const isEvicted = index === calculatedResults.length - 1 && resultsConfig.evictionPublished;
              return (
                <div key={res.id} className={`flex flex-col sm:flex-row items-center justify-between p-5 rounded-2xl border-2 transition duration-300 ${isEvicted ? 'bg-red-500/20 border-red-500' : 'bg-[#1A1A24] border-slate-800'} gap-4 shadow-xl`}>
                  <div className="flex items-center space-x-4">
                    <span className="w-10 h-10 rounded-full bg-[#D4AF37] text-black font-black text-base flex items-center justify-center shadow">
                      #{index + 1}
                    </span>
                    <img src={res.photo || 'https://placehold.co/200x200/0D0D11/D4AF37?text=Nominee'} alt={res.name} className="w-14 h-14 rounded-full object-cover border-2 border-[#D4AF37]" onError={(e)=>{e.target.src='https://placehold.co/200x200/0D0D11/D4AF37?text=Nominee'}} />
                    <div>
                      <h4 className="font-black text-white text-lg">{res.name}</h4>
                      <p className="text-xs font-mono text-slate-300">Save: {res.savePct.toFixed(2)}% | Evict: {res.evictPct.toFixed(2)}%</p>
                    </div>
                  </div>
                  <div className="flex items-center space-x-6">
                    <div className="text-right">
                      <div className="text-xs text-slate-400 font-black uppercase">Net Score</div>
                      <div className="text-xl font-black text-[#D4AF37] font-mono">{res.finalScore.toFixed(2)}</div>
                    </div>
                    {isEvicted && (
                      <span className="bg-red-600 text-white text-xs font-black px-4 py-2 rounded-xl uppercase tracking-wider animate-pulse shadow-lg">
                        EVICTED
                      </span>
                    )}
                  </div>
                </div>
              );
            })}
          </div>
        )}
      </div>
    </div>
  );
}

function AdminDashboard({ housemates, setHousemates, activeRound, setActiveRound, announcements, setAnnouncements, resultsConfig, setResultsConfig, votes, setVotes, currentUser, setCurrentUser, nominations, setNominations, weeklyStatuses, setWeeklyStatuses, archives, setArchives, syncToCloud, showAlert, goBack }) {
  const [adminPassword, setAdminPassword] = useState('');
  const [activeAdminTab, setActiveAdminTab] = useState('leaderboard'); 

  const [editingHmId, setEditingHmId] = useState(null);
  const [nameInput, setNameInput] = useState('');
  const [bioInput, setBioInput] = useState('');
  const [photoInput, setPhotoInput] = useState('');
  const [codeInput, setCodeInput] = useState('');

  // Nominee Custom Override State
  const [customNomineeName, setCustomNomineeName] = useState('');
  const [customNomineePhoto, setCustomNomineePhoto] = useState('');

  const handleAdminLogin = (e) => {
    e.preventDefault();
    if (adminPassword === '2009CAPYBARA18') {
      setCurrentUser({ role: 'admin' });
    } else {
      showAlert('Incorrect administrator password.');
    }
  };

  if (!currentUser || currentUser.role !== 'admin') {
    return (
      <div className="max-w-md mx-auto px-4 py-20 animate-fadeIn">
        <div className="bg-[#14141C] border-4 border-[#D4AF37] rounded-3xl p-8 shadow-2xl space-y-6">
          <div className="text-center space-y-3">
            <div className="w-14 h-14 rounded-full bg-[#D4AF37] flex items-center justify-center text-black mx-auto shadow-lg">
              <Lock className="w-7 h-7" />
            </div>
            <h2 className="text-2xl font-black text-white">ADMINISTRATOR LOGIN</h2>
            <p className="text-xs text-slate-300 font-medium">Restricted secure management gateway.</p>
          </div>

          <form onSubmit={handleAdminLogin} className="space-y-4">
            <div className="space-y-1">
              <label className="text-xs font-black uppercase text-[#D4AF37]">Admin Password</label>
              <input 
                type="password" 
                value={adminPassword} 
                onChange={e => setAdminPassword(e.target.value)} 
                placeholder="Enter password..." 
                className="w-full bg-[#1A1A24] border border-slate-800 rounded-xl px-4 py-3 text-sm text-white focus:outline-none focus:border-[#D4AF37] transition"
                required
              />
            </div>
            <button type="submit" className="w-full py-4 bg-[#D4AF37] text-black font-black rounded-xl hover:bg-amber-400 transition duration-300 shadow-lg transform active:scale-95">
              AUTHENTICATE ADMIN
            </button>
          </form>
        </div>
      </div>
    );
  }

  const weekKey = `week_${activeRound.week}`;
  const currentWeekNominations = nominations[weekKey] || {}; 

  const nominationTally = {};
  housemates.forEach(h => {
    nominationTally[h.id] = { id: h.id, name: h.name, points: 0, receivedFrom: [] };
  });

  Object.entries(currentWeekNominations).forEach(([nominatorId, sub]) => {
    const nominator = housemates.find(h => h.id === nominatorId);
    const nominatorName = nominator ? nominator.name : nominatorId;

    if (sub.first && nominationTally[sub.first.to]) {
      nominationTally[sub.first.to].points += sub.first.points;
      nominationTally[sub.first.to].receivedFrom.push({
        from: nominatorName,
        points: sub.first.points,
        reason: sub.first.reason
      });
    }
    if (sub.second && nominationTally[sub.second.to]) {
      nominationTally[sub.second.to].points += sub.second.points;
      nominationTally[sub.second.to].receivedFrom.push({
        from: nominatorName,
        points: sub.second.points,
        reason: sub.second.reason
      });
    }
  });

  const sortedLeaderboard = Object.values(nominationTally).sort((a, b) => b.points - a.points);

  const handleSaveHousemate = async (e) => {
    e.preventDefault();
    let updated;
    if (editingHmId) {
      updated = housemates.map(h => h.id === editingHmId ? { ...h, name: nameInput, bio: bioInput, photo: photoInput || h.photo, code: codeInput } : h);
      setEditingHmId(null);
      showAlert('Housemate updated and synced to cloud.');
    } else {
      const newId = `HM-00${housemates.length + 1}`;
      const newHm = {
        id: newId,
        name: nameInput,
        bio: bioInput,
        photo: photoInput || 'https://placehold.co/400x500/0D0D11/D4AF37?text=Housemate',
        status: 'Active',
        code: codeInput || '12345'
      };
      updated = [...housemates, newHm];
      showAlert(`Housemate ${newId} created and synced.`);
    }

    setHousemates(updated);
    await syncToCloud({ housemates: updated });

    setNameInput('');
    setBioInput('');
    setPhotoInput('');
    setCodeInput('');
  };

  const startEdit = (hm) => {
    setEditingHmId(hm.id);
    setNameInput(hm.name);
    setBioInput(hm.bio);
    setPhotoInput(hm.photo);
    setCodeInput(hm.code);
  };

  const handleStatusChange = async (id, newStatus) => {
    const updated = housemates.map(h => h.id === id ? { ...h, status: newStatus } : h);
    setHousemates(updated);

    let updatedStatuses = weeklyStatuses;
    if (newStatus === 'Evicted') {
      updatedStatuses = {
        ...weeklyStatuses,
        [id]: { ...weeklyStatuses[id], [activeRound.week]: 'Evicted' }
      };
      setWeeklyStatuses(updatedStatuses);
    }

    await syncToCloud({ housemates: updated, weeklyStatuses: updatedStatuses });
    showAlert(`Status updated to ${newStatus} and synced live to cloud.`);
  };

  const handleAddCustomNominee = async (e) => {
    e.preventDefault();
    if (!customNomineeName.trim()) {
      showAlert('Please enter a nominee name or duo title.');
      return;
    }

    const newNom = {
      id: `NOM-${Date.now()}`,
      label: customNomineeName.trim(),
      photo: customNomineePhoto.trim() || 'https://placehold.co/400x500/0D0D11/D4AF37?text=Nominee'
    };

    const updatedRound = {
      ...activeRound,
      nominees: [...(activeRound.nominees || []), newNom]
    };

    setActiveRound(updatedRound);
    await syncToCloud({ activeRound: updatedRound });

    setCustomNomineeName('');
    setCustomNomineePhoto('');
    showAlert(`Added "${newNom.label}" to official Week ${activeRound.week} nominees (Synced Live!).`);
  };

  const handleRemoveNominee = async (nomId) => {
    const updatedRound = {
      ...activeRound,
      nominees: (activeRound.nominees || []).filter(n => n.id !== nomId)
    };

    setActiveRound(updatedRound);
    await syncToCloud({ activeRound: updatedRound });
    showAlert('Nominee removed and synced to cloud.');
  };

  const handleStartNewWeek = async () => {
    const nextWeekNum = activeRound.week + 1;

    const newArchives = [
      { week: activeRound.week, title: `Week ${activeRound.week} Competition & Results`, summary: `Completed week ${activeRound.week} cycle with official nominees and eviction results.` },
      ...archives
    ];

    const newRound = {
      week: nextWeekNum,
      status: 'Voting Open',
      method: 'Online',
      nominees: []
    };

    const newVotes = {
      ...votes,
      [nextWeekNum]: { save: {}, evict: {} }
    };

    const newResultsConfig = { showPartial: true, showFinal: false, evictionPublished: false, votingManuallyClosed: false };

    setArchives(newArchives);
    setActiveRound(newRound);
    setVotes(newVotes);
    setResultsConfig(newResultsConfig);

    await syncToCloud({
      archives: newArchives,
      activeRound: newRound,
      votes: newVotes,
      resultsConfig: newResultsConfig
    });

    showAlert(`Week ${nextWeekNum} started! All devices reset to Week ${nextWeekNum} live.`);
  };

  return (
    <div className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8 py-12 space-y-8 animate-fadeIn">
      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between border-b border-slate-800 pb-6 gap-4">
        <div>
          <h1 className="text-3xl font-black text-white">ADMINISTRATOR CONTROL PANEL</h1>
          <p className="text-sm text-slate-300 font-medium">Manage housemates, duo nominees, leaderboards, and live cloud sync.</p>
        </div>
        <button 
          onClick={handleStartNewWeek}
          className="px-5 py-3 bg-emerald-500 hover:bg-emerald-400 text-black font-black text-xs rounded-xl shadow-xl flex items-center space-x-2 transition duration-300 transform active:scale-95"
        >
          <RotateCcw className="w-4 h-4" />
          <span>START NEXT WEEK (LIVE RESET CYCLE)</span>
        </button>
      </div>

      <div className="flex flex-wrap gap-2">
        {['leaderboard', 'housemates', 'rounds', 'voting', 'results', 'content'].map(tab => (
          <button
            key={tab}
            onClick={() => setActiveAdminTab(tab)}
            className={`px-4 py-2.5 text-xs font-black uppercase rounded-xl transition duration-300 transform active:scale-95 ${activeAdminTab === tab ? 'bg-[#D4AF37] text-black shadow-lg' : 'bg-[#1A1A24] text-white hover:bg-slate-800'}`}
          >
            {tab}
          </button>
        ))}
      </div>

      {activeAdminTab === 'leaderboard' && (
        <div className="bg-[#14141C] border-4 border-[#D4AF37] rounded-3xl p-6 sm:p-8 space-y-6 shadow-2xl animate-fadeIn">
          <div className="border-b border-slate-800 pb-4 flex items-center justify-between">
            <div>
              <h3 className="text-2xl font-black text-[#D4AF37]">CONFIDENTIAL NOMINATION LEADERBOARD (WEEK {activeRound.week})</h3>
              <p className="text-xs text-slate-300 font-medium">Aggregated housemate votes and private reasons.</p>
            </div>
            <span className="bg-red-500 text-white text-xs font-black px-3 py-1 rounded-full uppercase">Vault Secure</span>
          </div>

          <div className="space-y-6">
            {sortedLeaderboard.map((entry, index) => (
              <div key={entry.id} className="bg-[#1A1A24] border border-slate-800 rounded-2xl p-6 space-y-4 shadow-xl">
                <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-2 border-b border-slate-800 pb-3">
                  <div className="flex items-center space-x-3">
                    <span className="w-8 h-8 rounded-full bg-[#D4AF37] text-black font-black text-sm flex items-center justify-center">
                      #{index + 1}
                    </span>
                    <div>
                      <h4 className="text-lg font-black text-white">{entry.name}</h4>
                      <span className="text-xs font-mono text-[#D4AF37]">{entry.id}</span>
                    </div>
                  </div>
                  <div className="bg-[#0D0D11] px-4 py-2 rounded-xl border border-slate-800 text-right">
                    <span className="text-xs text-slate-400 font-bold uppercase">Total Points: </span>
                    <span className="text-xl font-black text-[#D4AF37] font-mono">{entry.points}</span>
                  </div>
                </div>

                {entry.receivedFrom.length === 0 ? (
                  <p className="text-xs text-slate-400 italic">No nomination points received this week.</p>
                ) : (
                  <div className="space-y-3 pt-2">
                    <h5 className="text-xs font-black uppercase text-slate-400">Nomination Ballots Received:</h5>
                    <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
                      {entry.receivedFrom.map((ballot, bIdx) => (
                        <div key={bIdx} className="bg-[#0D0D11] p-4 rounded-xl border border-slate-800 space-y-1">
                          <div className="flex justify-between text-xs font-bold text-[#D4AF37]">
                            <span>From: {ballot.from}</span>
                            <span>+{ballot.points} pts</span>
                          </div>
                          <p className="text-xs text-slate-300 italic">"{ballot.reason}"</p>
                        </div>
                      ))}
                    </div>
                  </div>
                )}
              </div>
            ))}
          </div>
        </div>
      )}

      {activeAdminTab === 'housemates' && (
        <div className="space-y-8 animate-fadeIn">
          <div className="bg-[#14141C] border-4 border-[#D4AF37] rounded-3xl p-6 sm:p-8 space-y-6 shadow-2xl">
            <h3 className="text-xl font-black text-[#D4AF37]">{editingHmId ? `Edit Housemate (${editingHmId})` : 'Add New Housemate'}</h3>
            <form onSubmit={handleSaveHousemate} className="grid grid-cols-1 sm:grid-cols-2 gap-4">
              <input 
                type="text" 
                value={nameInput} 
                onChange={e => setNameInput(e.target.value)} 
                placeholder="Full Name / Display Name" 
                className="bg-[#1A1A24] border border-slate-800 rounded-xl px-4 py-3 text-sm text-white focus:outline-none focus:border-[#D4AF37]"
                required
              />
              <input 
                type="text" 
                value={codeInput} 
                onChange={e => setCodeInput(e.target.value)} 
                placeholder="Access Code (e.g. ABC12)" 
                className="bg-[#1A1A24] border border-slate-800 rounded-xl px-4 py-3 text-sm text-white focus:outline-none focus:border-[#D4AF37]"
                required
              />
              <input 
                type="text" 
                value={photoInput} 
                onChange={e => setPhotoInput(e.target.value)} 
                placeholder="Image URL link (e.g. https://...)" 
                className="bg-[#1A1A24] border border-slate-800 rounded-xl px-4 py-3 text-sm text-white sm:col-span-2 focus:outline-none focus:border-[#D4AF37]"
              />
              <textarea 
                value={bioInput} 
                onChange={e => setBioInput(e.target.value)} 
                placeholder="Short biography..." 
                className="bg-[#1A1A24] border border-slate-800 rounded-xl p-4 text-sm text-white h-24 sm:col-span-2 focus:outline-none focus:border-[#D4AF37]"
                required
              />
              <div className="sm:col-span-2 flex space-x-3">
                <button type="submit" className="px-6 py-3.5 bg-[#D4AF37] text-black font-black rounded-xl hover:bg-amber-400 transition duration-300 shadow-lg transform active:scale-95">
                  {editingHmId ? 'Update & Sync Cloud' : 'Add & Sync Cloud'}
                </button>
                {editingHmId && (
                  <button type="button" onClick={() => { setEditingHmId(null); setNameInput(''); setBioInput(''); setPhotoInput(''); setCodeInput(''); }} className="px-6 py-3.5 bg-[#1A1A24] text-white rounded-xl hover:bg-slate-800 transition font-bold">
                    Cancel
                  </button>
                )}
              </div>
            </form>
          </div>

          <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-6">
            {housemates.map(hm => (
              <div key={hm.id} className="bg-[#14141C] border-2 border-slate-800 rounded-3xl p-6 space-y-4 shadow-xl flex flex-col justify-between">
                <div className="space-y-3">
                  <div className="flex items-center justify-between">
                    <span className="text-xs font-black text-black bg-[#D4AF37] px-3 py-1 rounded-full">{hm.id}</span>
                    <span className={`text-[10px] uppercase font-black px-2 py-0.5 rounded-full ${hm.status === 'Active' ? 'bg-emerald-500 text-black' : hm.status === 'Suspended' ? 'bg-amber-500 text-black' : 'bg-red-500 text-white'}`}>{hm.status}</span>
                  </div>
                  <img src={hm.photo} alt={hm.name} className="w-20 h-20 mx-auto rounded-full object-cover border-2 border-[#D4AF37]" />
                  <h4 className="font-black text-white text-lg text-center">{hm.name}</h4>
                  <p className="text-xs text-slate-300 line-clamp-2 font-medium">{hm.bio}</p>
                </div>
                <div className="pt-4 border-t border-slate-800 space-y-3">
                  <button onClick={() => startEdit(hm)} className="text-xs font-black text-[#D4AF37] hover:underline flex items-center space-x-1">
                    <Edit2 className="w-3.5 h-3.5" />
                    <span>Edit Profile</span>
                  </button>
                  <select 
                    value={hm.status}
                    onChange={(e) => handleStatusChange(hm.id, e.target.value)}
                    className="w-full bg-[#1A1A24] border border-slate-700 rounded-xl px-3 py-1.5 text-xs text-white font-bold focus:outline-none focus:border-[#D4AF37]"
                  >
                    <option value="Active">Active (Eligible)</option>
                    <option value="Suspended">Suspended (Revoked)</option>
                    <option value="Evicted">Evicted (Revoked)</option>
                  </select>
                </div>
              </div>
            ))}
          </div>
        </div>
      )}

      {activeAdminTab === 'rounds' && (
        <div className="bg-[#14141C] border-4 border-[#D4AF37] rounded-3xl p-6 sm:p-8 space-y-8 shadow-2xl animate-fadeIn">
          <div>
            <h3 className="text-xl font-black text-[#D4AF37] flex items-center space-x-2">
              <Sliders className="w-6 h-6 text-[#D4AF37]" />
              <span>MANAGE OFFICIAL NOMINEES & DUOS (WEEK {activeRound.week})</span>
            </h3>
            <p className="text-xs text-slate-300 mt-1">
              Add individual housemates, custom duo titles (e.g. "Aurelius & Beatrix"), or plot twist nominees directly to the cloud.
            </p>
          </div>

          <form onSubmit={handleAddCustomNominee} className="bg-[#1A1A24] p-6 rounded-2xl border border-slate-800 space-y-4">
            <h4 className="font-black text-white text-sm">Add Custom Nominee Entry</h4>
            <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
              <input 
                type="text" 
                value={customNomineeName} 
                onChange={e => setCustomNomineeName(e.target.value)} 
                placeholder="Nominee Name or Duo (e.g. 'Aurelius & Beatrix')" 
                className="bg-[#0D0D11] border border-slate-800 rounded-xl px-4 py-3 text-sm text-white font-bold focus:outline-none focus:border-[#D4AF37]"
                required
              />
              <input 
                type="text" 
                value={customNomineePhoto} 
                onChange={e => setCustomNomineePhoto(e.target.value)} 
                placeholder="Optional Photo URL..." 
                className="bg-[#0D0D11] border border-slate-800 rounded-xl px-4 py-3 text-sm text-white focus:outline-none focus:border-[#D4AF37]"
              />
            </div>
            <button type="submit" className="px-6 py-3 bg-[#D4AF37] text-black font-black rounded-xl hover:bg-amber-400 transition duration-300 shadow-lg text-xs uppercase">
              + ADD TO LIVE CLOUD NOMINEES
            </button>
          </form>

          <div className="space-y-4">
            <h4 className="font-black text-white text-sm uppercase">Confirmed Nominees for Week {activeRound.week}</h4>
            {(activeRound.nominees || []).length === 0 ? (
              <p className="text-xs text-slate-400 italic">No nominees confirmed for this week yet.</p>
            ) : (
              <div className="grid grid-cols-1 sm:grid-cols-2 md:grid-cols-3 gap-4">
                {(activeRound.nominees || []).map((nom, idx) => (
                  <div key={nom.id || idx} className="bg-[#1A1A24] border border-slate-800 rounded-2xl p-4 flex items-center justify-between shadow-lg">
                    <div className="flex items-center space-x-3">
                      <span className="w-8 h-8 rounded-full bg-[#D4AF37] text-black font-black text-sm flex items-center justify-center">
                        {String.fromCharCode(65 + idx)}
                      </span>
                      <div>
                        <h5 className="font-black text-white text-sm">{nom.label}</h5>
                        <span className="text-[10px] text-slate-400 font-mono">ID: {nom.id}</span>
                      </div>
                    </div>
                    <button 
                      onClick={() => handleRemoveNominee(nom.id)}
                      className="text-red-400 hover:text-red-300 p-2 rounded-lg hover:bg-[#0D0D11] transition"
                      title="Remove Nominee"
                    >
                      <Trash2 className="w-4 h-4" />
                    </button>
                  </div>
                ))}
              </div>
            )}
          </div>
        </div>
      )}

      {activeAdminTab === 'voting' && (
        <div className="bg-[#14141C] border-4 border-[#D4AF37] rounded-3xl p-6 sm:p-8 space-y-6 shadow-2xl animate-fadeIn">
          <div className="flex flex-col sm:flex-row sm:items-center justify-between border-b border-slate-800 pb-4 gap-4">
            <div>
              <h3 className="text-xl font-black text-[#D4AF37]">Voting Manual Control & Cloud Totals</h3>
              <p className="text-xs text-slate-300">Open or close public voting in real time across all devices.</p>
            </div>
            <button 
              onClick={async () => {
                const newState = !resultsConfig.votingManuallyClosed;
                const updatedConfig = { ...resultsConfig, votingManuallyClosed: newState };
                setResultsConfig(updatedConfig);
                await syncToCloud({ resultsConfig: updatedConfig });
                showAlert(`Voting manually ${newState ? 'CLOSED' : 'OPENED'} live for all users.`);
              }}
              className={`px-5 py-3 rounded-xl font-black text-xs uppercase shadow transition duration-300 transform active:scale-95 ${resultsConfig.votingManuallyClosed ? 'bg-emerald-500 text-black' : 'bg-red-600 text-white'}`}
            >
              {resultsConfig.votingManuallyClosed ? '🟢 OPEN LIVE VOTING NOW' : '🔒 CLOSE LIVE VOTING'}
            </button>
          </div>

          <div className="grid grid-cols-1 sm:grid-cols-2 gap-6 pt-2">
            <div className="bg-[#1A1A24] border border-slate-800 rounded-2xl p-6 space-y-4">
              <h4 className="font-black text-emerald-400 text-sm uppercase">Vote to Save Totals</h4>
              <div className="space-y-2 font-mono text-sm font-bold">
                {Object.entries(votes[activeRound.week]?.save || {}).map(([id, count]) => (
                  <div key={id} className="flex justify-between border-b border-slate-800 pb-2">
                    <span className="text-white">{id}</span>
                    <span className="text-emerald-400">{count} votes</span>
                  </div>
                ))}
              </div>
            </div>
            <div className="bg-[#1A1A24] border border-slate-800 rounded-2xl p-6 space-y-4">
              <h4 className="font-black text-red-400 text-sm uppercase">Vote to Evict Totals</h4>
              <div className="space-y-2 font-mono text-sm font-bold">
                {Object.entries(votes[activeRound.week]?.evict || {}).map(([id, count]) => (
                  <div key={id} className="flex justify-between border-b border-slate-800 pb-2">
                    <span className="text-white">{id}</span>
                    <span className="text-red-400">{count} votes</span>
                  </div>
                ))}
              </div>
            </div>
          </div>
        </div>
      )}

      {activeAdminTab === 'results' && (
        <div className="bg-[#14141C] border-4 border-[#D4AF37] rounded-3xl p-6 sm:p-8 space-y-6 shadow-2xl animate-fadeIn">
          <h3 className="text-xl font-black text-[#D4AF37]">Results Live Visibility Controls</h3>
          <div className="grid grid-cols-1 sm:grid-cols-3 gap-6">
            <div className="bg-[#1A1A24] p-6 rounded-2xl border border-slate-800 space-y-4 flex flex-col justify-between">
              <div>
                <h4 className="font-black text-white text-base">Partial Tally Graphic</h4>
                <p className="text-xs text-slate-300 mt-1 font-medium">Control live visibility of partial tally board.</p>
              </div>
              <button 
                onClick={async () => {
                  const updatedConfig = { ...resultsConfig, showPartial: !resultsConfig.showPartial };
                  setResultsConfig(updatedConfig);
                  await syncToCloud({ resultsConfig: updatedConfig });
                }}
                className={`w-full py-3 rounded-xl font-black text-xs uppercase shadow transition duration-300 transform active:scale-95 ${resultsConfig.showPartial ? 'bg-emerald-500 text-black' : 'bg-[#0D0D11] text-white border border-slate-700'}`}
              >
                {resultsConfig.showPartial ? 'Visible Live' : 'Hidden Live'}
              </button>
            </div>

            <div className="bg-[#1A1A24] p-6 rounded-2xl border border-slate-800 space-y-4 flex flex-col justify-between">
              <div>
                <h4 className="font-black text-white text-base">Final Results Graphic</h4>
                <p className="text-xs text-slate-300 mt-1 font-medium">Publish final scores live to all visitors.</p>
              </div>
              <button 
                onClick={async () => {
                  const updatedConfig = { ...resultsConfig, showFinal: !resultsConfig.showFinal };
                  setResultsConfig(updatedConfig);
                  await syncToCloud({ resultsConfig: updatedConfig });
                }}
                className={`w-full py-3 rounded-xl font-black text-xs uppercase shadow transition duration-300 transform active:scale-95 ${resultsConfig.showFinal ? 'bg-emerald-500 text-black' : 'bg-[#0D0D11] text-white border border-slate-700'}`}
              >
                {resultsConfig.showFinal ? 'Published Live' : 'Unpublished'}
              </button>
            </div>

            <div className="bg-[#1A1A24] p-6 rounded-2xl border border-slate-800 space-y-4 flex flex-col justify-between">
              <div>
                <h4 className="font-black text-white text-base">Eviction Announcement</h4>
                <p className="text-xs text-slate-300 mt-1 font-medium">Mark lowest-ranking nominee as evicted live.</p>
              </div>
              <button 
                onClick={async () => {
                  const updatedConfig = { ...resultsConfig, evictionPublished: !resultsConfig.evictionPublished };
                  setResultsConfig(updatedConfig);
                  await syncToCloud({ resultsConfig: updatedConfig });
                  showAlert(`Eviction publication status updated live on cloud.`);
                }}
                className={`w-full py-3 rounded-xl font-black text-xs uppercase shadow transition duration-300 transform active:scale-95 ${resultsConfig.evictionPublished ? 'bg-red-600 text-white' : 'bg-[#0D0D11] text-white border border-slate-700'}`}
              >
                {resultsConfig.evictionPublished ? 'Eviction Published Live' : 'Publish Eviction'}
              </button>
            </div>
          </div>
        </div>
      )}

      {activeAdminTab === 'content' && (
        <div className="bg-[#14141C] border-4 border-[#D4AF37] rounded-3xl p-6 sm:p-8 space-y-6 shadow-2xl animate-fadeIn">
          <h3 className="text-xl font-black text-[#D4AF37]">Broadcast Announcements Management</h3>
          <form onSubmit={async (e) => {
            e.preventDefault();
            const title = e.target.title.value;
            const content = e.target.content.value;
            const updatedAnn = [{ id: Date.now(), title, date: new Date().toISOString().split('T')[0], content }, ...announcements];
            setAnnouncements(updatedAnn);
            await syncToCloud({ announcements: updatedAnn });
            e.target.reset();
            showAlert('Announcement published and synced live.');
          }} className="space-y-4 bg-[#1A1A24] p-6 rounded-2xl border border-slate-800">
            <h4 className="font-black text-white text-sm">Post New Broadcast Update</h4>
            <input type="text" name="title" placeholder="Announcement Title" className="w-full bg-[#0D0D11] border border-slate-800 rounded-xl px-4 py-3 text-sm text-white font-bold" required />
            <textarea name="content" placeholder="Announcement content..." className="w-full bg-[#0D0D11] border border-slate-800 rounded-xl p-4 text-sm text-white h-28 font-medium" required />
            <button type="submit" className="px-6 py-3.5 bg-[#D4AF37] text-black font-black rounded-xl hover:bg-amber-400 transition duration-300 shadow-lg transform active:scale-95">
              PUBLISH LIVE ANNOUNCEMENT
            </button>
          </form>
        </div>
      )}
    </div>
  );
}
